<template>
    <div class="min-h-screen bg-slate-950 text-white font-sans p-8 flex flex-col items-center justify-center selection:bg-fuchsia-500">
      
      <div v-if="!isTournamentActive && !winner" class="w-full max-w-2xl bg-slate-900 border border-slate-800 rounded-2xl p-8 shadow-[0_0_40px_-10px_rgba(168,85,247,0.4)]">
        <h1 class="text-4xl font-black mb-6 text-transparent bg-clip-text bg-gradient-to-r from-fuchsia-400 to-cyan-400 uppercase tracking-widest">
          Enter The Arena
        </h1>
        <p class="text-slate-400 mb-4">List your options below, separated by commas or new lines.</p>
        
        <textarea 
          v-model="rawInput" 
          rows="5"
          class="w-full bg-slate-950 border border-slate-700 rounded-xl p-4 text-cyan-50 focus:outline-none focus:border-fuchsia-500 focus:ring-1 focus:ring-fuchsia-500 transition-all resize-none mb-6"
          placeholder="Code the API, Play Video Games, Sleep, Read a book..."
        ></textarea>
        
        <button 
          @click="startTournament"
          :disabled="parsedCandidates.length < 2"
          class="w-full py-4 rounded-xl font-bold text-lg uppercase tracking-wider transition-all duration-300 disabled:opacity-50 disabled:cursor-not-allowed bg-gradient-to-r from-fuchsia-600 to-purple-600 hover:from-fuchsia-500 hover:to-purple-500 shadow-[0_0_20px_rgba(192,38,211,0.5)] hover:shadow-[0_0_30px_rgba(192,38,211,0.8)]"
        >
          Initialize Bracket
        </button>
      </div>
  
      <div v-else-if="isTournamentActive && currentMatch.length === 2" class="w-full max-w-4xl flex flex-col items-center">
        <h2 class="text-2xl font-bold text-slate-400 mb-12 tracking-widest uppercase">
          Round {{ currentRoundNumber }} • Match {{ matchCount }}
        </h2>
        
        <div class="flex flex-col md:flex-row w-full items-center justify-center gap-8 md:gap-16 relative">
          
          <div class="absolute top-1/2 left-1/2 transform -translate-x-1/2 -translate-y-1/2 z-10 hidden md:flex items-center justify-center w-16 h-16 rounded-full bg-slate-950 border-2 border-slate-800 text-yellow-400 font-black text-xl shadow-[0_0_20px_rgba(250,204,21,0.3)]">
            VS
          </div>
  
          <button 
            @click="selectWinner(currentMatch[0])"
            class="group relative w-full md:w-1/2 p-1 rounded-2xl bg-gradient-to-br from-emerald-400 to-cyan-600 hover:scale-105 transition-transform duration-300 shadow-[0_0_30px_-5px_rgba(52,211,153,0.5)] cursor-pointer text-left"
          >
            <div class="bg-slate-950 h-full w-full rounded-xl p-8 flex items-center justify-center text-center group-hover:bg-opacity-80 transition-all">
              <span class="text-3xl font-bold text-emerald-300">{{ currentMatch[0] }}</span>
            </div>
          </button>
  
          <button 
            @click="selectWinner(currentMatch[1])"
            class="group relative w-full md:w-1/2 p-1 rounded-2xl bg-gradient-to-br from-rose-500 to-orange-500 hover:scale-105 transition-transform duration-300 shadow-[0_0_30px_-5px_rgba(244,63,94,0.5)] cursor-pointer text-left"
          >
            <div class="bg-slate-950 h-full w-full rounded-xl p-8 flex items-center justify-center text-center group-hover:bg-opacity-80 transition-all">
              <span class="text-3xl font-bold text-rose-300">{{ currentMatch[1] }}</span>
            </div>
          </button>
        </div>
        
        <div class="mt-12 text-slate-500 font-medium">
          Remaining Matches in Round: {{ upcomingMatches.length }}
        </div>
      </div>
  
      <div v-else-if="winner" class="w-full max-w-2xl text-center">
        <h3 class="text-xl text-slate-400 uppercase tracking-widest mb-4">Definitive Winner</h3>
        <div class="p-1 rounded-3xl bg-gradient-to-r from-yellow-300 via-amber-500 to-yellow-600 shadow-[0_0_60px_rgba(245,158,11,0.6)] animate-pulse mb-12">
          <div class="bg-slate-900 rounded-[23px] py-16 px-8">
            <span class="text-5xl font-black text-transparent bg-clip-text bg-gradient-to-r from-yellow-200 to-amber-500">
              {{ winner }}
            </span>
          </div>
        </div>
        
        <button 
          @click="reset"
          class="px-8 py-3 rounded-lg border border-slate-700 text-slate-300 hover:bg-slate-800 hover:text-white transition-colors"
        >
          Start New Bracket
        </button>
      </div>
  
    </div>
  </template>
  
  <script setup lang="ts">
  import { ref, computed } from 'vue'
  
  const rawInput = ref('')
  const isTournamentActive = ref(false)
  const candidates = ref<string[]>([])
  const currentRoundNumber = ref(1)
  const matchCount = ref(1)
  
  const upcomingMatches = ref<string[][]>([])
  const currentMatch = ref<string[]>([])
  const nextRoundCandidates = ref<string[]>([])
  const winner = ref<string | null>(null)
  
  const parsedCandidates = computed(() => {
    return rawInput.value
      .split(/[\n,]+/)
      .map(item => item.trim())
      .filter(item => item.length > 0)
  })
  
  const startTournament = () => {
    let list = [...parsedCandidates.value]
    for (let i = list.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [list[i], list[j]] = [list[j], list[i]];
    }
    
    candidates.value = list
    isTournamentActive.value = true
    currentRoundNumber.value = 1
    winner.value = null
    
    setupRound()
  }
  
  const setupRound = () => {
    upcomingMatches.value = []
    nextRoundCandidates.value = []
    matchCount.value = 1
    
    let currentPool = [...candidates.value]
    
    while (currentPool.length >= 2) {
      upcomingMatches.value.push([currentPool.shift()!, currentPool.shift()!])
    }
    
    if (currentPool.length === 1) {
      nextRoundCandidates.value.push(currentPool[0])
    }
    
    currentMatch.value = upcomingMatches.value.shift() || []
  }
  
  const selectWinner = (winningOption: string) => {
    nextRoundCandidates.value.push(winningOption)
    
    if (upcomingMatches.value.length > 0) {
      currentMatch.value = upcomingMatches.value.shift() || []
      matchCount.value++
    } else {
      candidates.value = [...nextRoundCandidates.value]
      
      if (candidates.value.length === 1) {
        winner.value = candidates.value[0]
        isTournamentActive.value = false
      } else {
        currentRoundNumber.value++
        setupRound()
      }
    }
  }
  
  const reset = () => {
    isTournamentActive.value = false
    winner.value = null
    rawInput.value = ''
    candidates.value = []
  }
  </script>
